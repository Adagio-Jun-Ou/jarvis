# Pinned runtime dependencies for the Alex agent endpoint.
# These are the versions the agent was tested against in dev. The bundle's
# job spec uses the same set; production images are built from this file.
#
# Why each pin matters:
#   mlflow            — ResponsesAgent (the serving contract, available
#                       since 3.0) plus tracing and the eval gate.
#   databricks-agents   — agents.deploy() helper
#   langchain-core      — OPTIONAL, for interop. agent/langchain_tools.py
#                       exports Alex's governed tools as LangChain tools;
#                       nothing in Alex's own loop imports it, and every
#                       import is lazy, so dropping it only disables the
#                       interop.
#   langgraph           — OPTIONAL, for interop. agent/langgraph_agent.py
#                       builds a ReAct graph over those governed tools.
#                       Alex's own loop is agent/vm.py and does not use it.
#   databricks-sdk      — WorkspaceClient for warehouse + serving
#   psycopg[binary]     — Lakebase (pgvector) connection for memory + KG
#   pydantic            — message schemas
#   httpx               — internal API client
#   cryptography        — browser cookie vault (Fernet), RS256 keys
#   playwright          — governed browser tool
#   PyJWT               — JWT verification (alex/identity.py)
#   numpy               — retrieval matrix math (agent/rag.py)
#
# Deliberately NOT here: databricks-langchain. It was pinned for a
# UCFunctionToolkit/ChatDatabricks path that nothing imports; Alex reaches
# Unity Catalog and Vector Search through its own tools. Shipping a
# dependency the code does not use is a supply-chain surface and it tells
# the next reader the system is built on a framework it is not.

databricks-agents==1.0.0

databricks-sdk==0.133.0
